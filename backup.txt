// Wlanie ekranu SSD1306 (np. modul 0.96" OLED I2C) na ESP32
//
// Wymagane biblioteki (Arduino IDE -> Sketch -> Include Library -> Manage Libraries):
//   - "Adafruit SSD1306"
//   - "Adafruit GFX Library"
//
// Typowe okablowanie modulu OLED:
//   VCC -> 3V3 (albo 5V, moduly maja regulator)
//   GND -> GND
//   SDA -> GPIO 21
//   SCL -> GPIO 22

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ---- Ustawienia ----
#define SCREEN_WIDTH   128   // OLED: 128 lub 64
#define SCREEN_HEIGHT  64    // OLED: 64 lub 32
#define OLED_ADDR      0x3C  // czesty adres I2C; bywa tez 0x3D
#define OLED_RESET     -1    // pin resetu podlaczony do RST ESP32, zwykle -1

#define I2C_SDA        21
#define I2C_SCL        22

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(115200);

  // Inicjalizacja magistrali I2C na wybranych pinach
  Wire.begin(I2C_SDA, I2C_SCL);

  // Sprawdzenie, czy ekran odpowiada
  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR)) {
    Serial.println(F("Blad: nie znaleziono SSD1306."));
    Serial.println(F("Sprawdz: zasilanie, kable SDA/SCL, adres I2C (0x3C/0x3D)."));
    while (true) {
      delay(1000);  // zatrzymaj program - mozesz odczytac komunikat w Monitorze Szeregowym
    }
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println(F("SSD1306 OK"));
  display.println(F(""));
  display.println(F("ESP32 + OLED I2C"));
  display.display();  // wyslanie bufora na ekran
  delay(1500);
}

void loop() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println(F("Ekran dziala!"));
  display.println(F(""));
  display.print(F("Czas [ms]: "));
  display.println(millis());

  // prosty pasek postepu
  int w = map((millis() / 50) % 100, 0, 99, 0, SCREEN_WIDTH - 2);
  display.drawRect(0, SCREEN_HEIGHT - 10, SCREEN_WIDTH - 1, 9, SSD1306_WHITE);
  display.fillRect(1, SCREEN_HEIGHT - 9, w, 7, SSD1306_WHITE);

  display.display();
  delay(50);
}
